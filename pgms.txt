PROGRAM 1 — ARRAY OPERATION

#include <stdio.h>

void main()
{
    int a[10], n, i, pos, item;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Array: ");
    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\nEnter position to insert: ");
    scanf("%d", &pos);

    printf("Enter element: ");
    scanf("%d", &item);

    for(i = n; i >= pos; i--)
        a[i] = a[i - 1];

    a[pos - 1] = item;
    n++;

    printf("After insertion: ");
    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\nEnter position to delete: ");
    scanf("%d", &pos);

    for(i = pos - 1; i < n - 1; i++)
        a[i] = a[i + 1];

    n--;

    printf("After deletion: ");
    for(i = 0; i < n; i++)
        printf("%d ", a[i]);
}


PROGRAM 2 — LINKED LIST SORTING

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *head = NULL;

void create()
{
    int n, i;
    struct node *newnode, *temp;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        newnode = malloc(sizeof(struct node));

        printf("Enter data: ");
        scanf("%d", &newnode->data);

        newnode->next = NULL;

        if(head == NULL)
            head = newnode;
        else
        {
            temp = head;

            while(temp->next != NULL)
                temp = temp->next;

            temp->next = newnode;
        }
    }
}

void sort()
{
    struct node *p, *q;
    int temp;

    for(p = head; p != NULL; p = p->next)
    {
        for(q = p->next; q != NULL; q = q->next)
        {
            if(p->data > q->data)
            {
                temp = p->data;
                p->data = q->data;
                q->data = temp;
            }
        }
    }
}

void display()
{
    struct node *temp = head;

    printf("Sorted list: ");

    while(temp != NULL)
    {
        printf("%d ", temp->data);
        temp = temp->next;
    }
}

void main()
{
    create();
    sort();
    display();
}

{

    for(i = 0; i < n; i++)
    {
        newnode = malloc(sizeof(struct node));

        printf("Enter data: ");
        scanf("%d", &newnode->data);

        newnode->next = NULL;

        if(head == NULL)
            head = newnode;
        else
        {
            temp = head;

            while(temp->next != NULL)
                temp = temp->next;

            temp->next = newnode;
        }
    }
}

void insertPosition()
{
    int pos, value, i;
    struct node *newnode, *temp;

    printf("Enter position: ");
    scanf("%d", &pos);

    printf("Enter value: ");
    scanf("%d", &value);

    newnode = malloc(sizeof(struct node));
    newnode->data = value;

    if(pos == 1)
    {
        newnode->next = head;
        head = newnode;
        return;
    }

    temp = head;

    for(i = 1; i < pos - 1 && temp != NULL; i++)
        temp = temp->next;

    if(temp == NULL)
        return;

    newnode->next = temp->next;
    temp->next = newnode;
}

void display()
{
    struct node *temp = head;

    printf("Linked list: ");

    while(temp != NULL)
    {
        printf("%d -> ", temp->data);
        temp = temp->next;
    }

    printf("NULL");
}

void main()
{
    create();
    insertPosition();
    display();
}



PROGRAM 4 — STACK USING LINKED LIST

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

void main()
{
    int ch, n, item, i;
    struct node *top = NULL, *temp;

    while(1)
    {
        printf("\n1. Push\n2. Pop\n3. Peek\n4. Display\n5. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        switch(ch)
        {
            case 1:
                printf("Enter number of elements: ");
                scanf("%d", &n);

                for(i = 1; i <= n; i++)
                {
                    temp = malloc(sizeof(struct node));

                    printf("Enter element: ");
                    scanf("%d", &item);

                    temp->data = item;
                    temp->next = top;
                    top = temp;
                }
                break;

            case 2:
                if(top == NULL)
                    printf("\nStack Underflow");
                else
                {
                    temp = top;
                    printf("\nDeleted element = %d", temp->data);
                    top = top->next;
                    free(temp);
                }
                break;

            case 3:
                if(top == NULL)
                    printf("\nStack is empty");
                else
                    printf("\nTop element = %d", top->data);
                break;

            case 4:
                if(top == NULL)
                    printf("\nStack is empty");
                else
                {
                    temp = top;
                    printf("\nElements: ");

                    while(temp != NULL)
                    {
                        printf("%d ", temp->data);
                        temp = temp->next;
                    }
                }
                break;

            case 5:
                return;

            default:
                printf("\nInvalid choice");
        }
    }
}


PROGRAM 5 — QUEUE USING ARRAY

#include <stdio.h>
#include <stdlib.h>

#define SIZE 20

void main()
{
    int queue[SIZE];
    int front = 0, rear = -1;
    int ch, value, i;

    do
    {
        printf("\n1. Enqueue");
        printf("\n2. Dequeue");
        printf("\n3. Display");
        printf("\n4. Exit");
        printf("\nEnter your choice: ");
        scanf("%d", &ch);

        switch(ch)
        {
            case 1:
                if(rear == SIZE - 1)
                    printf("\nQueue Overflow");
                else
                {
                    printf("Enter value: ");
                    scanf("%d", &value);
                    queue[++rear] = value;
                }
                break;

            case 2:
                if(front > rear)
                    printf("\nQueue Underflow");
                else
                    printf("\nDeleted = %d", queue[front++]);
                break;

            case 3:
                if(front > rear)
                    printf("\nQueue is empty");
                else
                {
                    printf("\nQueue: ");
                    for(i = front; i <= rear; i++)
                        printf("%d ", queue[i]);
                }
                break;

            case 4:
                printf("\nExit");
                break;

            default:
                printf("\nInvalid choice");
        }

    } while(ch != 4);
}


PROGRAM 6 — BINARY SEARCH

#include <stdio.h>
#include <conio.h>

void main()
{
    int a[50], n, key;
    int low, high, mid, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements in sorted order:\n");

    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter element to search: ");
    scanf("%d", &key);

    low = 0;
    high = n - 1;

    while(low <= high)
    {
        mid = (low + high) / 2;

        if(a[mid] == key)
        {
            printf("Element found at position %d", mid + 1);
            return;
        }
        else if(key < a[mid])
            high = mid - 1;
        else
            low = mid + 1;
    }

    printf("Element not found");

    getch();
}

PROGRAM 8 — BUBBLE SORT

#include <stdio.h>
#include <conio.h>

void main()
{
    int a[50], n, i, j, temp;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter the elements:\n");

    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - i - 1; j++)
        {
            if(a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("Sorted elements:\n");

    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    getch();
}
